      function buildReviewQuestion(item, practiceOnly=false) {
        const point=item.point, skill=item.skill;
        const skillState=pointState(point.id).skills[skill.id];
        const stage=Math.max(1,Number(skillState.stage||1));
        const contrastCases=skill.id==="contrast"?CONTRAST_CASES.filter(candidate=>candidate.pointId===point.id):[];
        const rotation=Math.max(0,Number(skillState.correct||0)+Number(skillState.incorrect||0));
        const contrastCase=contrastCases.length?contrastCases[rotation%contrastCases.length]:null;
        const examples=Array.isArray(point.examples)?point.examples:[];
        const example=contrastCase||examples[(rotation+SKILLS.findIndex(entry=>entry.id===skill.id))%Math.max(1,examples.length)];
        if(!example)return null;
        const accepted=contrastCase?[point.particle]:(Array.isArray(point.accepted)&&point.accepted.length?point.accepted:[point.particle]);
        const acceptedSet=new Set(accepted.map(normalizeAnswer));
        const nearby=contrastPoints(point).map(candidate=>candidate.particle);
        const pool=[...(contrastCase?.choices||[]),...(point.distractors||[]),...nearby];
        const alternatives=shuffle([...new Set(pool.filter(value=>value&&!acceptedSet.has(normalizeAnswer(value))))]).slice(0,3);
        const guided=stage<=2&&alternatives.length===3;
        return {
          point,skill,example,practiceOnly,contrastCase,
          type:guided?"choice-particle":"input",
          prompt:"Welche Partikel passt hier zur deutschen Aussage?",
          correct:point.particle,accepted,choices:guided?shuffle([point.particle,...alternatives]):[]
        };
      }
